file(GLOB ais4104_module_headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*/*.h")

set(ais4104_module_names "")
foreach (ais4104_module_header IN LISTS ais4104_module_headers)
    get_filename_component(ais4104_module_directory "${ais4104_module_header}" DIRECTORY)
    get_filename_component(ais4104_module_name "${ais4104_module_directory}" NAME)
    list(APPEND ais4104_module_names "${ais4104_module_name}")
endforeach ()
list(REMOVE_DUPLICATES ais4104_module_names)

foreach (ais4104_module_name IN LISTS ais4104_module_names)
    file(GLOB ais4104_module_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${ais4104_module_name}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/${ais4104_module_name}/*.cpp")
    message(STATUS "module: ${ais4104_module_name}")
    if (NOT TARGET praxis::${ais4104_module_name})
        message(FATAL_ERROR "modules/${ais4104_module_name} is named for the praxis extension praxis::${ais4104_module_name}, and no such target exists. A module folder takes the name of the praxis extension it binds")
    endif ()
    add_library(ais4104_modules_${ais4104_module_name} STATIC)
    add_library(ais4104::modules::${ais4104_module_name} ALIAS ais4104_modules_${ais4104_module_name})
    target_sources(ais4104_modules_${ais4104_module_name} PRIVATE ${ais4104_module_sources})
    target_include_directories(ais4104_modules_${ais4104_module_name} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(ais4104_modules_${ais4104_module_name} PUBLIC praxis::${ais4104_module_name})
    target_compile_features(ais4104_modules_${ais4104_module_name} PUBLIC cxx_std_20)
    ais4104_target_warnings(ais4104_modules_${ais4104_module_name})
endforeach ()
